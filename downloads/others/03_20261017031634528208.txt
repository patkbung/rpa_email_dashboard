import requests
from bs4 import BeautifulSoup
from openpyxl import Workbook
from openpyxl.styles import Font, PatternFill, Alignment
from openpyxl.utils import get_column_letter

URL = "https://realpython.github.io/fake-jobs/"
page = requests.get(URL)
soup = BeautifulSoup(page.content, "html.parser")
results = soup.find(id="ResultsContainer")

python_jobs = results.find_all(
    "h2", string=lambda text: "python" in text.lower()
)

python_job_cards = [
    h2.parent.parent.parent for h2 in python_jobs
]

print(f"พบงาน Python จำนวน: {len(python_job_cards)} ตำแหน่ง\n")

jobs = []
for job_card in python_job_cards:
    # class จริงจากเว็บ: "title is-5" / "subtitle is-6 company" / "location"
    title_el    = job_card.find("h2", class_="title")
    company_el  = job_card.find("h3", class_="company")
    location_el = job_card.find("p",  class_="location")
    link_tag    = job_card.find("a",  string="Apply")

    title    = title_el.text.strip()    if title_el    else "N/A"
    company  = company_el.text.strip()  if company_el  else "N/A"
    location = location_el.text.strip() if location_el else "N/A"
    link     = link_tag["href"]         if link_tag    else "N/A"

    print(f"Title   : {title}")
    print(f"Company : {company}")
    print(f"Location: {location}")
    print(f"Link    : {link}\n")

    jobs.append((title, company, location, link))

wb = Workbook()
ws = wb.active
ws.title = "Python Jobs"

headers = ["Title", "Company", "Location", "Apply Link"]
header_font  = Font(bold=True, color="FFFFFF", name="Arial", size=16)
header_fill  = PatternFill("solid", start_color="2563EB")
header_align = Alignment(horizontal="center", vertical="center")

for col, text in enumerate(headers, start=1):
    cell           = ws.cell(row=1, column=col, value=text)
    cell.font      = header_font
    cell.fill      = header_fill
    cell.alignment = header_align

ws.row_dimensions[1].height = 22

data_font = Font(name="Arial", size=10)
for row_idx, (title, company, location, link) in enumerate(jobs, start=2):
    ws.cell(row=row_idx, column=1, value=title).font    = data_font
    ws.cell(row=row_idx, column=2, value=company).font  = data_font
    ws.cell(row=row_idx, column=3, value=location).font = data_font
    ws.cell(row=row_idx, column=4, value=link).font     = data_font

col_widths = [35, 30, 25, 65]
for i, width in enumerate(col_widths, start=1):
    ws.column_dimensions[get_column_letter(i)].width = width

wb.save("python_jobs.xlsx")
print("บันทึกไฟล์ python_jobs.xlsx เรียบร้อยแล้ว ✓")
